// Real static handler; API forwarded to the journey server.
import { webHandler } from '../src/server/http';
import type { AppConfig } from '../src/server/config';
const handler = webHandler(
  async (request) => {
    const u = new URL(request.url);
    return fetch('http://127.0.0.1:3000' + u.pathname + u.search, {
      method: request.method,
      headers: { cookie: request.headers.get('cookie') ?? '' },
    });
  },
  { secureCookies: false } as AppConfig,
);
Bun.serve({
  port: 4175,
  hostname: '127.0.0.1',
  fetch: (request) => handler(request, '127.0.0.1'),
});
